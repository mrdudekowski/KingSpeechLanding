#### **JavaScript инициализация:**
```javascript
function init2GISMap() {
    if (typeof DG !== 'undefined') {
        // Создание интерактивной карты
        var map = DG.map('yandex-map', {
            center: [43.1056, 131.8735],
            zoom: 16
        });
        
        // Маркер с popup
        var marker = DG.marker([43.1056, 131.8735]).addTo(map);
        marker.bindPopup(/* HTML popup */);
        
        // Контролы
        DG.control.zoom({position: 'topright'}).addTo(map);
        DG.control.scale().addTo(map);
    } else {
        // Fallback на iframe
        setTimeout(/* retry logic */, 2000);
    }
}
```
